cmake_minimum_required(VERSION 3.10)
project(u_connect_client_examples C ASM)
string(TOLOWER ${CMAKE_SYSTEM_NAME} OS_NAME)

# Enable ccache if available
find_program(CCACHE_PROGRAM ccache)
if(CCACHE_PROGRAM)
    message(STATUS "Found ccache: ${CCACHE_PROGRAM}")
    set(CMAKE_C_COMPILER_LAUNCHER ${CCACHE_PROGRAM})
    set(CMAKE_CXX_COMPILER_LAUNCHER ${CCACHE_PROGRAM})
endif()

# Include u-connectClient library definitions
include(../u-connectClient.cmake)

# Set binary output directory to bin/
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_CURRENT_SOURCE_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_CURRENT_SOURCE_DIR}/bin)

# Common source files for examples
set(EXAMPLE_COMMON_SRC
  example_utils.c
  ${U_CONNECT_CLIENT_UCX_API_SRC}
  ${U_CONNECT_CLIENT_AT_API_SRC}
)

# Option to build STM32 examples
option(BUILD_STM32_EXAMPLES "Build examples for STM32" OFF)

# Option to silence the AT command TX/RX trace (kept separate from U_CX_LOG_DEBUG)
option(DISABLE_AT_LOG "Disable AT command TX/RX logging" OFF)

# Option to silence [DBG] channel logging (app progress messages, BIN headers, etc.)
option(DISABLE_DBG_LOG "Disable DBG channel logging" OFF)

if(BUILD_STM32_EXAMPLES)
    # STM32 build path - requires cross-compilation toolchain
    include(../cmake/stm32.cmake)

    # Add STM32 examples
    add_stm32_example(NAME http_example_stm32 SOURCES http_example.c md5.c)
    add_stm32_example(NAME fw_upgrade_example_stm32 SOURCES fw_upgrade_example.c ../src/u_cx_xmodem.c)
    add_stm32_example(NAME ble_scan_example_stm32 SOURCES ble_scan_example.c)
    add_stm32_example(NAME ble_advertise_example_stm32 SOURCES ble_advertise_example.c)
    add_stm32_example(NAME wifi_scan_example_stm32 SOURCES wifi_scan_example.c)
    add_stm32_example(NAME wifi_ap_example_stm32 SOURCES wifi_ap_example.c)
    add_stm32_example(NAME socket_example_stm32 SOURCES socket_example.c)
    add_stm32_example(NAME uart_bridge_example_stm32 SOURCES uart_bridge_example.c)

    # Set U_CX_XMODEM_FILE_SUPPORT for firmware upgrade
    target_compile_definitions(fw_upgrade_example_stm32.elf PRIVATE U_CX_XMODEM_FILE_SUPPORT=1)

else()
    # Native build path (Linux/Windows/macOS)
    find_package(Threads REQUIRED)

    # Platform-specific port layer files
    if(WIN32)
      set(PORT_OS_SRC ../ports/os/u_port_windows.c)
      set(PORT_UART_SRC ../ports/uart/u_port_uart_windows.c)
      set(PORT_DEFINES -DU_PORT_WINDOWS)
      set(PORT_LIBS setupapi)  # u_port_uart_windows.c uses SetupDi* for FTDI latency timer / diagnostics
      if(MSVC)
        set(EXAMPLE_COMPILE_OPTIONS /J /WX /wd4200 /wd4996)
      else()
        # GCC/Clang on Windows (e.g. MSYS2/MinGW via Ninja)
        set(EXAMPLE_COMPILE_OPTIONS -Wall -Wextra -Werror -Wconversion -Wsign-conversion -Wshadow -pedantic)
      endif()
    else()
      set(PORT_OS_SRC ../ports/os/u_port_posix.c)
      set(PORT_UART_SRC ../ports/uart/u_port_uart_linux.c)
      set(PORT_DEFINES -DU_PORT_POSIX)
      set(PORT_LIBS Threads::Threads)
      set(EXAMPLE_COMPILE_OPTIONS -Wall -Wextra -Werror -Wconversion -Wsign-conversion -Wshadow -pedantic)
    endif()

    if(DISABLE_DBG_LOG)
      list(APPEND EXAMPLE_COMPILE_OPTIONS -DU_CX_LOG_DEBUG=0)
    else()
      list(APPEND EXAMPLE_COMPILE_OPTIONS -DU_CX_LOG_DEBUG=1)
    endif()
    if(DISABLE_AT_LOG)
      list(APPEND EXAMPLE_COMPILE_OPTIONS -DU_CX_LOG_AT=0)
    endif()

    # HTTP Example (OS-based)
    # md5.c: MD5 verification of downloaded files - native hosts only, not built for STM32
    add_executable(http_example
      http_example.c
      md5.c
      ${PORT_OS_SRC}
      ${PORT_UART_SRC}
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(http_example PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES})
    target_include_directories(http_example PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(http_example PRIVATE ${PORT_LIBS})
    endif()

    # HTTP Example No-OS
    add_executable(http_example_no_os
      http_example.c
      md5.c
      ../ports/os/u_port_no_os.c
      ${PORT_UART_SRC}
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(http_example_no_os PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES} -DU_PORT_NO_OS)
    target_include_directories(http_example_no_os PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(http_example_no_os PRIVATE ${PORT_LIBS})
    endif()

    # Firmware Upgrade Example (OS-based)
    add_executable(fw_upgrade_example
      fw_upgrade_example.c
      ${PORT_OS_SRC}
      ${PORT_UART_SRC}
      ../src/u_cx_xmodem.c
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(fw_upgrade_example PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES} -DU_CX_XMODEM_FILE_SUPPORT=1)
    target_include_directories(fw_upgrade_example PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(fw_upgrade_example PRIVATE ${PORT_LIBS})
    endif()

    # Firmware Upgrade Example No-OS
    add_executable(fw_upgrade_example_no_os
      fw_upgrade_example.c
      ../ports/os/u_port_no_os.c
      ${PORT_UART_SRC}
      ../src/u_cx_xmodem.c
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(fw_upgrade_example_no_os PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES} -DU_PORT_NO_OS -DU_CX_XMODEM_FILE_SUPPORT=1)
    target_include_directories(fw_upgrade_example_no_os PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(fw_upgrade_example_no_os PRIVATE ${PORT_LIBS})
    endif()

    # BLE Scan Example (OS-based)
    add_executable(ble_scan_example
      ble_scan_example.c
      ${PORT_OS_SRC}
      ${PORT_UART_SRC}
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(ble_scan_example PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES})
    target_include_directories(ble_scan_example PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(ble_scan_example PRIVATE ${PORT_LIBS})
    endif()

    # BLE Advertise Example (OS-based)
    add_executable(ble_advertise_example
      ble_advertise_example.c
      ${PORT_OS_SRC}
      ${PORT_UART_SRC}
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(ble_advertise_example PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES})
    target_include_directories(ble_advertise_example PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(ble_advertise_example PRIVATE ${PORT_LIBS})
    endif()

    # WiFi Scan Example (OS-based)
    add_executable(wifi_scan_example
      wifi_scan_example.c
      ${PORT_OS_SRC}
      ${PORT_UART_SRC}
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(wifi_scan_example PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES})
    target_include_directories(wifi_scan_example PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(wifi_scan_example PRIVATE ${PORT_LIBS})
    endif()

    # WiFi AP Example (OS-based)
    add_executable(wifi_ap_example
      wifi_ap_example.c
      ${PORT_OS_SRC}
      ${PORT_UART_SRC}
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(wifi_ap_example PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES})
    target_include_directories(wifi_ap_example PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(wifi_ap_example PRIVATE ${PORT_LIBS})
    endif()

    # Socket Example (OS-based)
    add_executable(socket_example
      socket_example.c
      ${PORT_OS_SRC}
      ${PORT_UART_SRC}
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(socket_example PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES})
    target_include_directories(socket_example PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(socket_example PRIVATE ${PORT_LIBS})
    endif()

    # MQTT <-> BLE Gateway Example (OS-based)
    add_executable(ble_mqtt_gateway_example
      ble_mqtt_gateway_example.c
      ${PORT_OS_SRC}
      ${PORT_UART_SRC}
      ${EXAMPLE_COMMON_SRC}
    )
    target_compile_options(ble_mqtt_gateway_example PRIVATE ${EXAMPLE_COMPILE_OPTIONS} ${PORT_DEFINES})
    target_include_directories(ble_mqtt_gateway_example PUBLIC ${U_CONNECT_CLIENT_INC} ${U_CONNECT_CLIENT_PORT_DIR})
    if(PORT_LIBS)
      target_link_libraries(ble_mqtt_gateway_example PRIVATE ${PORT_LIBS})
    endif()
endif()
